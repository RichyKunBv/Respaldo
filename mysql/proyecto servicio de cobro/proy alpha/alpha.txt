create database serviziogas;
use serviziogas;

CREATE TABLE Proveedores (
    id_proveedor INT UNSIGNED NOT NULL AUTO_INCREMENT,
    nombre_comercial VARCHAR(100) NOT NULL,
    nombre_contacto VARCHAR(150),
    telefono VARCHAR(20),
    plataformas_comunicacion VARCHAR(255), -- Para guardar 'WhatsApp, SMS', etc. [cite: 25]
    PRIMARY KEY (id_proveedor)
);

CREATE TABLE Inquilinos (
    id_inquilino INT UNSIGNED NOT NULL AUTO_INCREMENT,
    nombre_completo VARCHAR(150) NOT NULL,
    PRIMARY KEY (id_inquilino)
);

CREATE TABLE Departamentos (
    id_departamento INT UNSIGNED NOT NULL AUTO_INCREMENT,
    numero_departamento VARCHAR(10) NOT NULL UNIQUE, -- ej: '1', '2', 'A', 'B' [cite: 13]
    id_inquilino_responsable INT UNSIGNED,
    estado ENUM('Ocupado', 'Vacio') NOT NULL DEFAULT 'Ocupado', -- Para saber si está habitado [cite: 30]
    servicio_suspendido BOOLEAN NOT NULL DEFAULT FALSE, -- Para marcar si se cortó el servicio [cite: 28]
    PRIMARY KEY (id_departamento),
    FOREIGN KEY (id_inquilino_responsable) REFERENCES Inquilinos(id_inquilino)
);

CREATE TABLE Lecturas (
    id_lectura INT UNSIGNED NOT NULL AUTO_INCREMENT,
    id_departamento INT UNSIGNED NOT NULL,
    fecha_lectura DATE NOT NULL, -- Fecha en que se toma la lectura [cite: 14]
    lectura_medidor DECIMAL(10, 2) NOT NULL, -- El valor que marca el medidor
    consumo_mes_actual DECIMAL(10, 2) NOT NULL, -- La diferencia calculada del consumo [cite: 19, 20]
    precio_litro_gas DECIMAL(10, 2) NOT NULL, -- El costo del litro para ese periodo [cite: 16, 18]
    monto_a_pagar DECIMAL(10, 2) NOT NULL, -- El total a pagar (consumo * precio)
    PRIMARY KEY (id_lectura),
    FOREIGN KEY (id_departamento) REFERENCES Departamentos(id_departamento)
);

CREATE TABLE Pagos (
    id_pago INT UNSIGNED NOT NULL AUTO_INCREMENT,
    id_lectura INT UNSIGNED NOT NULL, -- Vincula el pago a un registro de consumo
    id_inquilino_paga INT UNSIGNED, -- Quién realizó el pago [cite: 26]
    fecha_pago DATE, -- La fecha en que se realizó el pago [cite: 23]
    fecha_limite_pago DATE NOT NULL, -- La fecha límite para pagar [cite: 21]
    monto_pagado DECIMAL(10, 2),
    metodo_pago ENUM('Efectivo', 'Transferencia', 'Cheque'), -- [cite: 22]
    estado_pago ENUM('Pagado', 'Pendiente', 'Demora', 'Retraso Severo') NOT NULL DEFAULT 'Pendiente', -- [cite: 27]
    PRIMARY KEY (id_pago),
    FOREIGN KEY (id_lectura) REFERENCES Lecturas(id_lectura),
    FOREIGN KEY (id_inquilino_paga) REFERENCES Inquilinos(id_inquilino)
);

commit;
